#include <stdio.h>

int main()
{
    int numeros[8];
    int pares[8];
    int impares[8];
    int i, cantidadPares = 0, cantidadImpares = 0;

    for(i = 0; i < 8; i++)
    {
        printf("Ingresa el numero %d: ", i + 1);
        scanf("%d", &numeros[i]);
    }

    printf("\nNumeros ingresados:\n");

    for(i = 0; i < 8; i++)
    {
        printf("%d ", numeros[i]);

        if(numeros[i] % 2 == 0)
        {
            pares[cantidadPares] = numeros[i];
            cantidadPares++;
        }
        else
        {
            impares[cantidadImpares] = numeros[i];
            cantidadImpares++;
        }
    }

    printf("\n\nNumeros pares:\n");

    for(i = 0; i < cantidadPares; i++)
    {
        printf("%d ", pares[i]);
    }

    printf("\n\nNumeros impares:\n");

    for(i = 0; i < cantidadImpares; i++)
    {
        printf("%d ", impares[i]);
    }

    printf("\n");

    return 0;
}
